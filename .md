This breakdown covers the dynamic configuration system and the self-managed local API handler. It uses a flexible mapping system so `host=local` or `host=default` translates to the correct internal URLs while allowing any custom URL.

### 1. Enhanced Settings Manager
This class handles the logic of translating keywords (`local`, `default`) into functional URLs and persisting them to a JSON file.

```python
import json
from pathlib import Path

CONFIG_DIR = Path.home() / ".pyanimecli"
CONFIG_FILE = CONFIG_DIR / "config.json"

# Static mappings for keyword hosts
HOST_MAP = {
    "default": "https://yumaapi.vercel.app",
    "local": "http://localhost:8192" # The local API usually runs on 3000
}

DEFAULT_SETTINGS = {
    "host": HOST_MAP["default"],
    "source": "sub",
    "proxy_url": "https://gammam3u8proxy-fxsb.vercel.app/cors?url=",
    "player": "vlc",
    "auto_update": True,
    "download_path": str(Path.home() / "Downloads" / "Anime")
}

class Settings:
    def __init__(self):
        CONFIG_DIR.mkdir(exist_ok=True)
        self.data = self.load()

    def load(self):
        if CONFIG_FILE.exists():
            with open(CONFIG_FILE, "r") as f:
                return {**DEFAULT_SETTINGS, **json.load(f)}
        return DEFAULT_SETTINGS.copy()

    def save(self):
        with open(CONFIG_FILE, "w") as f:
            json.dump(self.data, f, indent=4)

    def update(self, key, value):
        # Handle keywords
        if key == "host":
            if value.lower() in HOST_MAP:
                self.data[key] = HOST_MAP[value.lower()]
            else:
                self.data[key] = value # Custom URL
        elif value.lower() == "default" and key in DEFAULT_SETTINGS:
            self.data[key] = DEFAULT_SETTINGS[key]
        else:
            self.data[key] = value
        self.save()
```

### 2. Local API Automator (The `host=local` Logic)
This handles the lifecycle of the `YumaAnimeAPI`: downloading, updating based on GitHub's latest commit, and headless execution.

```python
import zipfile
import io
import time
import subprocess

class LocalServerManager:
    REPO_ZIP = "https://github.com/princevegetadev/YumaAnimeAPI/archive/refs/heads/main.zip"
    SERVER_DIR = CONFIG_DIR / "server_files"
    
    def get_latest_commit_date(self):
        try:
            # Check GitHub API for last commit date
            api_url = "https://api.github.com/repos/princevegetadev/YumaAnimeAPI/commits/main"
            res = requests.get(api_url, timeout=5)
            return res.json()['commit']['committer']['date']
        except:
            return None

    def needs_update(self):
        latest = self.get_latest_commit_date()
        status_file = self.SERVER_DIR / ".latest"
        if not status_file.exists() or not latest:
            return True
        return status_file.read_text().strip() != latest

    def install_or_update(self):
        console.print("[cyan]Updating local API environment...[/cyan]")
        r = requests.get(self.REPO_ZIP)
        with zipfile.ZipFile(io.BytesIO(r.content)) as z:
            z.extractall(self.SERVER_DIR)
        
        # Determine the extracted folder name
        extracted_folder = next(self.SERVER_DIR.glob("YumaAnimeAPI-*"))
        
        # Install deps
        subprocess.run([sys.executable, "-m", "pip", "install", "-r", str(extracted_folder / "requirements.txt")], 
                       stdout=subprocess.DEVNULL)
        
        # Record latest update
        latest_date = self.get_latest_commit_date()
        (self.SERVER_DIR / ".latest").write_text(latest_date or str(time.time()))

    def start_headless(self):
        try:
            # Simple check if something is already on port 3000
            requests.get("http://localhost:3000", timeout=1)
            return
        except:
            pass

        if self.needs_update():
            self.install_or_update()

        extracted_folder = next(self.SERVER_DIR.glob("YumaAnimeAPI-*"))
        console.print("[green]Launching Local API Server...[/green]")
        
        # Headless background process
        subprocess.Popen(
            [sys.executable, "app.py"],
            cwd=str(extracted_folder),
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            creationflags=subprocess.CREATE_NO_WINDOW if platform.system() == "Windows" else 0
        )
        time.sleep(3) # Let it warm up
```

### 3. Integrating with `main` and `make_request`
You need to modify the entry points so that settings are respected and the local server auto-starts if selected.

**Inside `make_request`:**
```python
settings = Settings()

def make_request(endpoint, params=None):
    # If host is set to the local URL, ensure server is up
    if "localhost:3000" in settings.data['host']:
        LocalServerManager().start_headless()
        
    url = f"{settings.data['host']}/{endpoint}"
    # ... use settings.data['proxy_url'] for stream links ...
```

**Inside `main()`'s Argument Parser:**
```python
# Add to argparse
parser.add_argument('--settings', nargs='*', help='Open settings menu or set key="value"')

# Handle the logic
if args.settings is not None:
    if len(args.settings) == 0:
        # INTERACTIVE TUI
        console.print(Panel("[bold cyan]pyanimecli Configuration[/bold cyan]"))
        for k, v in settings.data.items():
            console.print(f"[bold]{k}[/bold]: [yellow]{v}[/yellow]")
        
        key = input("\nEnter setting to change (or 'exit'): ").strip()
        if key in settings.data:
            val = input(f"Enter new value for {key}: ").strip()
            settings.update(key, val)
            console.print("[green]Settings saved.[/green]")
    else:
        # COMMAND LINE UPDATE (e.g., --settings host=local source=dub)
        for arg in args.settings:
            if "=" in arg:
                k, v = arg.split("=", 1)
                settings.update(k.strip(), v.strip())
                console.print(f"[green]Set {k} to {v}[/green]")
    sys.exit(0)
```

### 4. Advanced Settings Ideas
Since you wanted "much more," here are keys you should add to your `DEFAULT_SETTINGS` and logic:

*   **`player_args`**: Custom flags for VLC (e.g. `--network-caching=5000`).
*   **`force_proxy`**: (True/False) Forces the CORS proxy even for local hosts.
*   **`discord_rpc`**: Toggle for Rich Presence.
*   **`check_api_health`**: If set to true, `make_request` will automatically try the `default` host if the `custom` host fails, or vice-versa.
*   **`stream_type`**: (hls/dash/mp4) Preference for stream formats if available.
*   **`history`**: (True/False) Locally log what you've watched for a "Continue Watching" feature.
